Demo 2 — Counter (onclick + if/else + DOM)

Interaction Counter

Each click increments a let variable, updates the page, and fires a GA4 event. When the count crosses a threshold, the message changes — and a different GA4 event fires. 

let clickCount = 0;

function handleClick() {
  clickCount++;

  // Update the page
  document.getElementById('counter-display').textContent = clickCount;

  // Fire a GA4 event on every click
  gtag('event', 'button_click', {'count': clickCount});

  // Conditional: change message at threshold
  if (clickCount >= 10) {
    document.getElementById('counter-message').textContent = "You're dedicated!";
    gtag('event', 'milestone_reached', {'milestone': '10_clicks'});
  } else if (clickCount >= 5) {
    document.getElementById('counter-message').textContent = "Keep going...";
  } else {
    document.getElementById('counter-message').textContent = "Just getting started.";
  }
}

Demo 3 — Array → DOM (for loop + createElement)

Skills Rendered from a JavaScript Array

These skill tags are not written in HTML. A JavaScript array + a for loop builds each one and appends it to the page. Clicking the button also fires a GA4 event. 

const skills = ['HTML', 'CSS', 'JavaScript', 'GA4', 'Flexbox'];

function renderSkills() {
  const list = document.getElementById('skills-output');
  list.innerHTML = '';  // clear first

  for (let i = 0; i < skills.length; i++) {
    const item = document.createElement('li');
    item.textContent = skills[i];
    list.appendChild(item);
  }

  gtag('event', 'skills_rendered', {'count': skills.length});
  document.getElementById('skills-count').textContent = skills.length + ' skills loaded.';
}

    Demo 4 — Interest Filter (switch + array of objects)

    Filter by Category — Your Final Project Feature

    This is exactly what you will build for the final. A dropdown triggers a switch statement that filters an array of objects and re-renders the page. Each filter selection fires a GA4 event. 

    const items = [
  { name: 'Portfolio Site', category: 'Web',  description: 'HTML & CSS project' },
  { name: 'Python Script',  category: 'Code', description: 'Automates file sorting' },
  // ... more items
];

function filterItems(category) {
  switch (category) {
    case '':
      renderItems(items); break;
    case 'Web':
      renderItems(items.filter(i => i.category === 'Web')); break;
    // ... more cases
    default:
      renderItems(items);
  }
  gtag('event', 'filter_used', {'category': category || 'all'});
}
